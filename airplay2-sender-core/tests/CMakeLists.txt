function(sender_test target label)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE ../include ../core ../ptp "${CMAKE_BINARY_DIR}")
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE CONFIG_GCRYPT=1 HAVE_CLOCK_GETTIME=1 PACKAGE_NAME="airplay2-sender-core")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
    target_link_libraries(${target} PRIVATE airplay2_sender_core sender_platform PkgConfig::PROTOCOL)
    string(REGEX REPLACE "^test_" "" test_name "${target}")
    airplay2_register_test(${label}.${test_name} ${label} ${target})
endfunction()
