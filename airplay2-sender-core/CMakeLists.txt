cmake_minimum_required(VERSION 3.25)
project(airplay2_sender_core LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(AIRPLAY2_BUILD_PROTOCOL "Build the extracted sender protocol" ON)
option(AIRPLAY2_BUILD_SHARED "Build the explicit C ABI shared sender alongside the static sender" OFF)
if(WIN32)
    if(NOT CMAKE_HOST_WIN32 OR CMAKE_CROSSCOMPILING OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL "13.2.0")
        message(FATAL_ERROR "The verified Windows build uses native MinGW-w64 GCC 13.2.0 with MSVCRT")
    endif()
    include(platform/windows/dependencies.cmake)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET libevent libsodium)
endif()
if(NOT WIN32)
    find_package(Threads REQUIRED)
endif()
if(WIN32)
    set(platform_backend windows)
else()
    set(platform_backend posix)
    pkg_check_modules(PLATFORM_UUID REQUIRED IMPORTED_TARGET uuid)
endif()
add_library(sender_platform STATIC platform/common.c platform/network.c platform/${platform_backend}/backend.c
    platform/${platform_backend}/network.c platform/${platform_backend}/interfaces.c platform/${platform_backend}/clock.c)
target_include_directories(sender_platform PUBLIC platform)
target_compile_definitions(sender_platform PRIVATE _GNU_SOURCE)
target_link_libraries(sender_platform PUBLIC PkgConfig::PLATFORM)
if(NOT WIN32)
    target_link_libraries(sender_platform PRIVATE PkgConfig::PLATFORM_UUID Threads::Threads)
endif()
if(WIN32)
    target_compile_definitions(sender_platform PUBLIC _WIN32_WINNT=0x0602 SODIUM_STATIC __USE_MINGW_ANSI_STDIO=1)
    target_sources(sender_platform PRIVATE platform/windows/events.c)
    target_link_libraries(sender_platform PUBLIC ws2_32 iphlpapi bcrypt)
    target_link_options(sender_platform PUBLIC -static-libgcc)
endif()
if(AIRPLAY2_BUILD_PROTOCOL)
    if(NOT WIN32)
        pkg_check_modules(PROTOCOL REQUIRED IMPORTED_TARGET
            libevent libgcrypt gpg-error libsodium libplist-2.0 libavcodec libavutil uuid)
    endif()
    if(PROTOCOL_libavcodec_VERSION VERSION_GREATER_EQUAL "59")
        message(FATAL_ERROR "Use the frozen FFmpeg 4.4 reference dependency for this extraction")
    endif()
    add_library(airplay2_sender_core STATIC
        core/capability.c core/runtime.c core/target_probe.c core/session.c core/protocol_util.c
        core/codec.c core/crypto/chacha.c core/rtsp.c core/rtp.c core/airplay_events.c core/ptpd.c
        core/pairing/pair.c core/pairing/pair_homekit.c core/pairing/pair_fruit.c core/pairing/pair-tlv.c)
    target_sources(airplay2_sender_core PRIVATE ptp/src/airptp.c ptp/src/daemon.c ptp/src/ptp_msg_handle.c ptp/src/utils.c ptp/platform_adapter.c)
    target_include_directories(airplay2_sender_core PUBLIC include PRIVATE core ptp "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(airplay2_sender_core PRIVATE _GNU_SOURCE CONFIG_GCRYPT=1
        HAVE_CLOCK_GETTIME=1 PACKAGE_NAME="airplay2-sender-core")
    target_compile_options(airplay2_sender_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
    set_target_properties(airplay2_sender_core PROPERTIES C_EXTENSIONS ON)
    target_link_libraries(airplay2_sender_core PRIVATE sender_platform PkgConfig::PROTOCOL)
    if(WIN32)
        target_link_options(airplay2_sender_core INTERFACE -static)
    endif()
    if(NOT WIN32)
        target_link_libraries(airplay2_sender_core PRIVATE Threads::Threads m rt)
    endif()
    configure_file(core/config.h.in config.h)
    if(AIRPLAY2_BUILD_SHARED)
        get_target_property(sender_sources airplay2_sender_core SOURCES)
        get_target_property(sender_includes airplay2_sender_core INCLUDE_DIRECTORIES)
        get_target_property(sender_definitions airplay2_sender_core COMPILE_DEFINITIONS)
        get_target_property(sender_compile_options airplay2_sender_core COMPILE_OPTIONS)
        add_library(airplay2_sender_shared SHARED ${sender_sources})
        target_include_directories(airplay2_sender_shared PUBLIC include PRIVATE ${sender_includes})
        target_compile_definitions(airplay2_sender_shared PRIVATE ${sender_definitions} SENDER_SHARED_BUILD=1)
        target_compile_options(airplay2_sender_shared PRIVATE ${sender_compile_options})
        target_link_libraries(airplay2_sender_shared PRIVATE sender_platform PkgConfig::PROTOCOL)
        set_target_properties(sender_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
        set_target_properties(airplay2_sender_shared PROPERTIES C_EXTENSIONS ON PREFIX "" OUTPUT_NAME "airplay2_sender")
        if(WIN32)
            target_link_options(airplay2_sender_shared PRIVATE -static-libgcc -Wl,--exclude-all-symbols)
            # The tested archive explicitly exports its own API through COFF directives.
            # Strip only those directives from a private build copy, keeping code and the tested input intact.
            if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/libplist-internal.a" OR
               "${dependency_plist-2.0}" IS_NEWER_THAN "${CMAKE_CURRENT_BINARY_DIR}/libplist-internal.a")
                execute_process(COMMAND "${CMAKE_OBJCOPY}" --remove-section .drectve
                    "${dependency_plist-2.0}" "${CMAKE_CURRENT_BINARY_DIR}/libplist-internal.a" COMMAND_ERROR_IS_FATAL ANY)
            endif()
            target_link_options(airplay2_sender_shared PRIVATE "$<$<CONFIG:Release>:-Wl,--strip-debug>")
            get_target_property(shared_dependencies PkgConfig::PROTOCOL INTERFACE_LINK_LIBRARIES)
            list(REMOVE_ITEM shared_dependencies "${dependency_plist-2.0}")
            list(APPEND shared_dependencies "${CMAKE_CURRENT_BINARY_DIR}/libplist-internal.a")
            set_property(TARGET airplay2_sender_shared PROPERTY LINK_LIBRARIES "sender_platform;${shared_dependencies}")
        else()
            target_link_libraries(airplay2_sender_shared PRIVATE Threads::Threads m rt)
            set_target_properties(airplay2_sender_shared PROPERTIES C_VISIBILITY_PRESET hidden)
        endif()
    endif()
else()
    add_library(airplay2_sender_core INTERFACE)
    target_include_directories(airplay2_sender_core INTERFACE include)
endif()

option(BUILD_TESTING "Build behavioral tests" ON)
if(BUILD_TESTING)
    enable_testing()
    add_subdirectory(tests)
endif()
